# Root build file for FrostIDE. Configure through CMakePresets.json
# (win-clangcl, linux-clang and their sanitizer variants); the presets select
# the pinned Clang toolchain, vcpkg and Qt from sdk/sdk.lock.json.

cmake_minimum_required(VERSION 4.0)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk/sdk.lock.json" _frost_lock)
string(JSON FROST_QT_VERSION GET "${_frost_lock}" qt version)

project(FrostIDE VERSION 0.0.1 DESCRIPTION "FrostIDE C/C++ IDE" LANGUAGES C CXX)

if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "FrostIDE's own code builds with Clang only (PLAN.md D2); got ${CMAKE_CXX_COMPILER_ID}")
endif()

set(FROST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

# Named modules are banned in v1 (PLAN.md section 2.1), so skip CMake's C++20
# module dependency scan; it would run clang-scan-deps over every file.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
# Debug information embedded in objects (/Z7) in every configuration, so each object
# is self-contained and cacheable by sccache (FrostCompilerCache.cmake). clang-cl
# embeds it for /Zi too and never writes a compile-time PDB, but /Z7 says so
# explicitly; with cl.exe, /Zi's shared PDB would make compiles uncacheable.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# sccache as the compiler launcher (FROST_SCCACHE); must come before the first target.
include(FrostCompilerCache)

option(FROST_BUILD_TESTS "Build the test suites" ON)
option(FROST_BUILD_UI "Build Qt-based targets (src/ui, src/app)" ON)
set(FROST_SANITIZE "" CACHE STRING "Sanitizers to enable: address;undefined, thread or memory")
# Fuzz trees (presets win-clangcl-fuzz, linux-clang-fuzz; cmake/FrostFuzz.cmake, ADR 0013). -Full
# runs each target for FROST_FUZZ_SECONDS = 60; longer runs (600 s for the M1 exit, 1800 s for a
# future -Nightly) use separate build trees, so the -Full tree's cache keeps 60.
option(FROST_FUZZ "Build libFuzzer targets instead of corpus replays (needs ASan, no tests, no UI)" OFF)
set(FROST_FUZZ_SECONDS 60 CACHE STRING "Seconds each CTest fuzz.<name> run fuzzes")

if(FROST_FUZZ)
    if(NOT "address" IN_LIST FROST_SANITIZE)
        message(FATAL_ERROR "FROST_FUZZ needs AddressSanitizer: FROST_SANITIZE must contain address")
    endif()
    if(FROST_BUILD_UI)
        message(FATAL_ERROR "FROST_FUZZ needs FROST_BUILD_UI=OFF (fuzz trees are Qt-free)")
    endif()
    if(FROST_BUILD_TESTS)
        # gtest from the /MD triplet cannot link into /MT objects; normal trees build the replays.
        message(FATAL_ERROR "FROST_FUZZ needs FROST_BUILD_TESTS=OFF (normal trees build the corpus replays)")
    endif()
    if(MSVC AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreaded")
        # The pinned clang_rt.fuzzer-x86_64.lib records RuntimeLibrary=MT_StaticRelease and
        # _ITERATOR_DEBUG_LEVEL=0 (RelWithDebInfo matches); lld-link's /failifmismatch rejects
        # any other CRT. Its STL imports are defined by the pinned 14.44 libcpmt (ADR 0013).
        message(FATAL_ERROR "FROST_FUZZ under clang-cl needs CMAKE_MSVC_RUNTIME_LIBRARY=MultiThreaded "
                            "(got '${CMAKE_MSVC_RUNTIME_LIBRARY}')")
    endif()
    add_compile_options(-fsanitize=fuzzer-no-link)
    if(MSVC)
        # The same library records annotate_string=0 and annotate_vector=0 while ASan TUs record 1,
        # which fails /failifmismatch; fuzz trees lose std::string/vector container-overflow checks
        # (the Windows ASan tree keeps them; ADR 0013).
        add_compile_definitions(_DISABLE_STRING_ANNOTATION _DISABLE_VECTOR_ANNOTATION)
    endif()
    message(STATUS "FrostIDE fuzz targets: ${FROST_FUZZ_SECONDS} s per CTest run")
endif()

include(FrostCompilerOptions)
include(FrostLayers)

if(WIN32)
    # The application manifest shared by frost.exe and the Windows test executables.
    set(FROST_MANIFEST_PATH "${CMAKE_BINARY_DIR}/generated/frost.manifest")
    configure_file(cmake/windows-frost.manifest.in "${FROST_MANIFEST_PATH}" @ONLY)
    # file(CONFIGURE) rewrites the file only when its content changes, so a
    # reconfigure does not relink every test. The .rc files name the manifest in a
    # resource statement, which the resource compiler's dependency output does not
    # report, so each .rc source also gets OBJECT_DEPENDS on the manifest
    # (frost_add_test and src/app/CMakeLists.txt).
    set(FROST_TEST_MANIFEST_RC "${CMAKE_BINARY_DIR}/generated/test_manifest.rc")
    file(CONFIGURE OUTPUT "${FROST_TEST_MANIFEST_RC}"
        CONTENT "// Generated by CMakeLists.txt: embeds the FrostIDE manifest into a test executable.\n1 24 \"@FROST_MANIFEST_PATH@\"\n"
        @ONLY)
endif()

# Sanitizers apply to every FrostIDE target so libraries and tests agree on the runtime.
# frost_apply_compile_options (cmake/FrostCompilerOptions.cmake), which every target's helper
# calls, adds these options per target; frost.exe's Windows launcher alone opts out
# (NO_SANITIZE, ADR 0012).
set(FROST_SANITIZE_COMPILE_OPTIONS "")
set(FROST_SANITIZE_LINK_OPTIONS "")
if(FROST_SANITIZE)
    list(JOIN FROST_SANITIZE "," _frost_sanitizers)
    if(MSVC)
        # clang-cl: AddressSanitizer only, with the release CRT (/MD; see the
        # win-clangcl-asan preset). clang-cl refuses -fsanitize=address with /MDd, so the ASan
        # tree's Debug configuration does not build; use its RelWithDebInfo (or Release). If
        # UBSan is ever wanted here, use it together with ASan or in trap mode
        # (-fsanitize-trap=undefined): its standalone runtime is static-only and fails the
        # link of /MD code (measured 2026-11-01, ADR 0012). CMake links with lld-link
        # directly, so the ASan runtime inputs come from the toolchain
        # (FROST_ASAN_LINK_OPTIONS); lld-link would ignore -fsanitize.
        if(NOT FROST_SANITIZE STREQUAL "address")
            message(FATAL_ERROR "clang-cl builds support FROST_SANITIZE=address only (got '${FROST_SANITIZE}')")
        endif()
        # /Oy- is clang-cl's spelling of -fno-omit-frame-pointer (keeps ASan stacks complete).
        set(FROST_SANITIZE_COMPILE_OPTIONS -fsanitize=address /Oy-)
        # The static CRT (/MT, the fuzz trees) takes the static runtime thunk; /MD the dynamic one.
        if(CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreaded")
            set(FROST_SANITIZE_LINK_OPTIONS ${FROST_ASAN_STATIC_LINK_OPTIONS})
        else()
            set(FROST_SANITIZE_LINK_OPTIONS ${FROST_ASAN_LINK_OPTIONS})
        endif()
        # The ASan runtime DLL, copied next to every executable by frost_copy_runtime_dlls,
        # and loaded first by frost.exe's launcher (src/app/CMakeLists.txt).
        set(FROST_ASAN_RUNTIME_DLL "${FROST_ASAN_RUNTIME_DIR}/clang_rt.asan_dynamic-x86_64.dll")
        if(NOT EXISTS "${FROST_ASAN_RUNTIME_DLL}")
            message(FATAL_ERROR "ASan runtime not found: ${FROST_ASAN_RUNTIME_DLL}")
        endif()
    else()
        set(FROST_SANITIZE_COMPILE_OPTIONS
            -fsanitize=${_frost_sanitizers} -fno-omit-frame-pointer -fno-sanitize-recover=all)
        set(FROST_SANITIZE_LINK_OPTIONS -fsanitize=${_frost_sanitizers})
    endif()
    message(STATUS "FrostIDE sanitizers: ${_frost_sanitizers}")
endif()

if(FROST_BUILD_UI)
    find_package(Qt6 ${FROST_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Widgets)
    # Qt's plugin directory, used by tests to load the offscreen platform plugin, and its
    # DLL directory, from which frost_process_tests takes DLLs frost does not deploy.
    get_target_property(_frost_qmake Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(_frost_qt_prefix "${_frost_qmake}/../.." ABSOLUTE)
    set(FROST_QT_PLUGIN_DIR "${_frost_qt_prefix}/plugins")
    set(FROST_QT_BIN_DIR "${_frost_qt_prefix}/bin")
endif()

# Threads::Threads (frost::sched). Found here so the imported target is visible from
# the top-level directory, where FrostLayers resolves namespaced dependencies.
find_package(Threads REQUIRED)
# SQLite and xxHash (frost::storage, ADR 0014), found here for the same reason: FrostLayers' deferred
# graph check runs in the top-level directory and must resolve unofficial::sqlite3::sqlite3 and
# xxHash::xxhash. Fuzz trees install no vcpkg ports and build no storage, hence the guard
# (an undefined FROST_FUZZ is false). pugixml (frost::project's XML reader, ADR 0033) likewise.
if(NOT FROST_FUZZ)
    find_package(unofficial-sqlite3 CONFIG REQUIRED)
    find_package(xxHash CONFIG REQUIRED)
    find_package(pugixml CONFIG REQUIRED)
endif()
# asio::asio (header-only, BSL-1.0; frost::io's reactor, ADR 0020), found here for the same reason.
# Fuzz trees install no vcpkg ports (VCPKG_MANIFEST_INSTALL=OFF) but share the normal tree's
# installed directory and triplet, so they find the headers the normal configure installed.
find_package(asio CONFIG)
if(NOT asio_FOUND)
    if(FROST_FUZZ)
        message(FATAL_ERROR "asio not found: fuzz trees use the headers the normal preset installs "
                            "(configure win-clangcl or linux-clang first; ADR 0020)")
    endif()
    message(FATAL_ERROR "asio not found: add it to vcpkg.json and configure through a preset (ADR 0020)")
endif()

if(NOT FROST_FUZZ)
    # frost::diag's third-party libraries (ADR 0022). Fuzz trees skip frost::diag and install no
    # vcpkg manifest.
    find_package(crashpad CONFIG REQUIRED)
    find_package(cpptrace CONFIG REQUIRED)
    find_package(spdlog CONFIG REQUIRED)
endif()

if(FROST_BUILD_TESTS OR FROST_FUZZ)
    enable_testing()
endif()
if(FROST_BUILD_TESTS)
    find_package(GTest CONFIG REQUIRED)
    include(GoogleTest)
endif()
# RE2 and utf8proc (frost::text_editor, ADR 0025), found here so that FrostLayers' deferred graph
# check in the top-level directory resolves re2::re2 and utf8proc::utf8proc. Fuzz trees install no
# vcpkg ports and build no editor model, hence the guard (an undefined FROST_FUZZ is false).
if(NOT FROST_FUZZ)
    find_package(re2 CONFIG REQUIRED)
    find_package(utf8proc CONFIG REQUIRED)
endif()
include(FrostFuzz)

add_subdirectory(src)
if(FROST_BUILD_UI)
    # Build-time tools and the resources they compile into frost's output directory (ADR 0015).
    add_subdirectory(tools/theme_compiler)
    add_subdirectory(tools/icon_baker)
    add_subdirectory(resources)
endif()
if(FROST_BUILD_TESTS)
    add_subdirectory(tests)
elseif(FROST_FUZZ)
    add_subdirectory(tests/fuzz)
endif()
