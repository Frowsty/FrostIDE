# Root build file for FrostIDE. Configure through CMakePresets.json
# (win-clangcl, linux-clang and their sanitizer variants); the presets select
# the pinned Clang toolchain, vcpkg and Qt from sdk/sdk.lock.json.

cmake_minimum_required(VERSION 4.0)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk/sdk.lock.json" _frost_lock)
string(JSON FROST_QT_VERSION GET "${_frost_lock}" qt version)

project(FrostIDE VERSION 0.0.1 DESCRIPTION "FrostIDE C/C++ IDE" LANGUAGES C CXX)

if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "FrostIDE's own code builds with Clang only (PLAN.md D2); got ${CMAKE_CXX_COMPILER_ID}")
endif()

set(FROST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

# Named modules are banned in v1 (PLAN.md section 2.1), so skip CMake's C++20
# module dependency scan; it would run clang-scan-deps over every file.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
# Debug information embedded in objects (/Z7) keeps sccache effective on Windows.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

option(FROST_BUILD_TESTS "Build the test suites" ON)
option(FROST_BUILD_UI "Build Qt-based targets (src/ui, src/app)" ON)
set(FROST_SANITIZE "" CACHE STRING "Sanitizers to enable: address;undefined, thread or memory")

include(FrostCompilerOptions)
include(FrostLayers)

if(WIN32)
    # The application manifest shared by frost.exe and the Windows test executables.
    set(FROST_MANIFEST_PATH "${CMAKE_BINARY_DIR}/generated/frost.manifest")
    configure_file(cmake/windows-frost.manifest.in "${FROST_MANIFEST_PATH}" @ONLY)
    set(FROST_TEST_MANIFEST_RC "${CMAKE_BINARY_DIR}/generated/test_manifest.rc")
    file(WRITE "${FROST_TEST_MANIFEST_RC}"
        "// Generated by CMakeLists.txt: embeds the FrostIDE manifest into a test executable.\n"
        "1 24 \"${FROST_MANIFEST_PATH}\"\n")
endif()

# Sanitizers apply to every target so libraries and tests agree on the runtime.
if(FROST_SANITIZE)
    list(JOIN FROST_SANITIZE "," _frost_sanitizers)
    if(MSVC)
        # clang-cl: ASan links the dynamic runtime that matches /MD.
        add_compile_options(-fsanitize=${_frost_sanitizers} -fno-omit-frame-pointer)
        add_link_options(-fsanitize=${_frost_sanitizers})
    else()
        add_compile_options(-fsanitize=${_frost_sanitizers} -fno-omit-frame-pointer -fno-sanitize-recover=all)
        add_link_options(-fsanitize=${_frost_sanitizers})
    endif()
    message(STATUS "FrostIDE sanitizers: ${_frost_sanitizers}")
endif()

if(FROST_BUILD_UI)
    find_package(Qt6 ${FROST_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Widgets)
    # Qt's plugin directory, used by tests to load the offscreen platform plugin.
    get_target_property(_frost_qmake Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(_frost_qt_prefix "${_frost_qmake}/../.." ABSOLUTE)
    set(FROST_QT_PLUGIN_DIR "${_frost_qt_prefix}/plugins")
endif()

if(FROST_BUILD_TESTS)
    enable_testing()
    find_package(GTest CONFIG REQUIRED)
    include(GoogleTest)
endif()

add_subdirectory(src)
if(FROST_BUILD_TESTS)
    add_subdirectory(tests)
endif()
